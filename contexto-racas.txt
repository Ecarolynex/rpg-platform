import { ATTRIBUTE_CONFIG, type BaseAttributeKey } from "./personaRules";

export interface RaceTraitDefinition {
  id: string;
  nome: string;
  descricao: string;
}

export interface RaceDefinition {
  id: string;
  nome: string;
  /** Nomes antigos ou grafias alternativas ("Meio-orc"): personagens guardam a raça pelo nome. */
  aliases: string[];
  descricao: string;
  atributoBonus: Record<BaseAttributeKey, number>;
  hpBonus: number;
  mpBonus: number;
  tracos: RaceTraitDefinition[];
}

const ATRIBUTOS: BaseAttributeKey[] = ATTRIBUTE_CONFIG.map(({ key }) => key);

function bonusZerado(): Record<BaseAttributeKey, number> {
  return Object.fromEntries(ATRIBUTOS.map((key) => [key, 0])) as Record<BaseAttributeKey, number>;
}

function racaPadrao(
  id: string,
  nome: string,
  opcoes: {
    aliases?: string[];
    atributos?: Partial<Record<BaseAttributeKey, number>>;
    hp?: number;
    mp?: number;
    traco?: { nome: string; descricao: string };
  } = {},
): RaceDefinition {
  return {
    id,
    nome,
    aliases: opcoes.aliases ?? [],
    descricao: "",
    atributoBonus: { ...bonusZerado(), ...opcoes.atributos },
    hpBonus: opcoes.hp ?? 0,
    mpBonus: opcoes.mp ?? 0,
    tracos: opcoes.traco
      ? [{ id: `traco-${id}-1`, nome: opcoes.traco.nome, descricao: opcoes.traco.descricao }]
      : [],
  };
}

/** Ignora acentos, maiúsculas e espaços nas pontas ("Meio-orc" = "meio-Orc", "Anão" = "ANAO"). */
export function chaveDeRaca(nome: string | null | undefined): string {
  return (nome ?? "")
    .normalize("NFD")
    .replace(/[\u0300-\u036f]/g, "")
    .trim()
    .toLocaleLowerCase();
}

export function findRaceDefinition(
  nome: string | null | undefined,
  catalogo: RaceDefinition[] = DEFAULT_RACE_CATALOG,
): RaceDefinition | undefined {
  const chave = chaveDeRaca(nome);
  if (!chave) return undefined;
  return catalogo.find(
    (raca) =>
      chaveDeRaca(raca.nome) === chave
      || raca.aliases.some((alias) => chaveDeRaca(alias) === chave),
  );
}

function textoSeguro(valor: unknown): string {
  return typeof valor === "string" ? valor.trim().slice(0, 500) : "";
}

function numeroSeguro(valor: unknown, fallback = 0): number {
  return typeof valor === "number" && Number.isFinite(valor) ? Math.trunc(valor) : fallback;
}

export function normalizarCatalogoRacas(valor: unknown): RaceDefinition[] {
  if (!Array.isArray(valor)) return DEFAULT_RACE_CATALOG;

  return valor.flatMap((item): RaceDefinition[] => {
    if (typeof item !== "object" || item === null) return [];
    const entrada = item as Record<string, unknown>;
    const id = textoSeguro(entrada.id);
    const nome = textoSeguro(entrada.nome);
    if (!id || !nome) return [];

    const bonusEntrada =
      typeof entrada.atributoBonus === "object" && entrada.atributoBonus !== null
        ? (entrada.atributoBonus as Record<string, unknown>)
        : {};
    const atributoBonus = Object.fromEntries(
      ATRIBUTOS.map((atributo) => [atributo, numeroSeguro(bonusEntrada[atributo])]),
    ) as Record<BaseAttributeKey, number>;

    const tracos: RaceTraitDefinition[] = Array.isArray(entrada.tracos)
      ? entrada.tracos.flatMap((itemTraco): RaceTraitDefinition[] => {
          if (typeof itemTraco !== "object" || itemTraco === null) return [];
          const traco = itemTraco as Record<string, unknown>;
          const tracoId = textoSeguro(traco.id);
          const nomeTraco = textoSeguro(traco.nome);
          if (!tracoId || !nomeTraco) return [];
          return [{ id: tracoId, nome: nomeTraco, descricao: textoSeguro(traco.descricao) }];
        })
      : [];

    return [{
      id,
      nome,
      aliases: Array.isArray(entrada.aliases)
        ? [...new Set(entrada.aliases.map(textoSeguro).filter(Boolean))]
        : [],
      descricao: textoSeguro(entrada.descricao),
      atributoBonus,
      hpBonus: numeroSeguro(entrada.hpBonus),
      mpBonus: numeroSeguro(entrada.mpBonus),
      tracos,
    }];
  });
}

// ---------------------------------------------------------------------------
// Raças por campanha: o Mestre libera/bloqueia raças do catálogo global,
// ajusta bônus só para a mesa e cria raças exclusivas da campanha.
// ---------------------------------------------------------------------------

export interface CampaignRaceOverride {
  /** Impede escolher a raça em NOVOS personagens. Fichas existentes não mudam. */
  bloqueada?: boolean;
  /** Bônus de atributo que substituem os do catálogo global (só nesta campanha). */
  atributoBonus?: Partial<Record<BaseAttributeKey, number>>;
  hpBonus?: number;
  mpBonus?: number;
}

export interface CampaignRaceConfig {
  /** Chave = id da raça (estável mesmo se o nome mudar). */
  overrides: Record<string, CampaignRaceOverride>;
  exclusivas: RaceDefinition[];
}

export const EMPTY_CAMPAIGN_RACE_CONFIG: CampaignRaceConfig = { overrides: {}, exclusivas: [] };

export interface RaceResolvida extends RaceDefinition {
  origem: "GLOBAL" | "CAMPANHA";
  bloqueada: boolean;
  /** true quando o Mestre trocou os bônus do catálogo global nesta campanha. */
  personalizada: boolean;
}

function overrideTemConteudo(o: CampaignRaceOverride): boolean {
  return (
    Boolean(o.bloqueada)
    || o.atributoBonus !== undefined
    || o.hpBonus !== undefined
    || o.mpBonus !== undefined
  );
}

export function normalizarConfigRacasCampanha(valor: unknown): CampaignRaceConfig {
  if (typeof valor !== "object" || valor === null) return EMPTY_CAMPAIGN_RACE_CONFIG;
  const entrada = valor as Record<string, unknown>;

  const overrides: Record<string, CampaignRaceOverride> = {};
  if (typeof entrada.overrides === "object" && entrada.overrides !== null) {
    for (const [id, bruto] of Object.entries(entrada.overrides)) {
      if (typeof bruto !== "object" || bruto === null) continue;
      const o = bruto as Record<string, unknown>;
      const resultado: CampaignRaceOverride = {};

      if (o.bloqueada === true) resultado.bloqueada = true;

      if (typeof o.atributoBonus === "object" && o.atributoBonus !== null) {
        const bonus = o.atributoBonus as Record<string, unknown>;
        const parcial: Partial<Record<BaseAttributeKey, number>> = {};
        for (const atributo of ATRIBUTOS) {
          const v = bonus[atributo];
          if (typeof v === "number" && Number.isFinite(v)) parcial[atributo] = Math.trunc(v);
        }
        if (Object.keys(parcial).length > 0) resultado.atributoBonus = parcial;
      }

      if (typeof o.hpBonus === "number" && Number.isFinite(o.hpBonus)) {
        resultado.hpBonus = Math.trunc(o.hpBonus);
      }
      if (typeof o.mpBonus === "number" && Number.isFinite(o.mpBonus)) {
        resultado.mpBonus = Math.trunc(o.mpBonus);
      }

      if (overrideTemConteudo(resultado)) overrides[id] = resultado;
    }
  }

  const exclusivas = Array.isArray(entrada.exclusivas)
    ? normalizarCatalogoRacas(entrada.exclusivas)
    : [];

  return { overrides, exclusivas };
}

/**
 * Raças válidas numa campanha: catálogo global (com os ajustes do Mestre) + raças exclusivas.
 * Raças bloqueadas continuam na lista (com bloqueada = true) para que fichas existentes
 * mantenham os bônus; quem monta o seletor de criação é que filtra as bloqueadas.
 */
export function racasDaCampanha(
  catalogo: RaceDefinition[],
  config: CampaignRaceConfig = EMPTY_CAMPAIGN_RACE_CONFIG,
): RaceResolvida[] {
  const globais: RaceResolvida[] = catalogo.map((raca) => {
    const o = config.overrides[raca.id];
    return {
      ...raca,
      atributoBonus: o?.atributoBonus
        ? { ...raca.atributoBonus, ...o.atributoBonus }
        : raca.atributoBonus,
      hpBonus: o?.hpBonus ?? raca.hpBonus,
      mpBonus: o?.mpBonus ?? raca.mpBonus,
      origem: "GLOBAL",
      bloqueada: Boolean(o?.bloqueada),
      personalizada:
        o?.atributoBonus !== undefined || o?.hpBonus !== undefined || o?.mpBonus !== undefined,
    };
  });

  const nomesGlobais = new Set(catalogo.map((raca) => chaveDeRaca(raca.nome)));
  const exclusivas: RaceResolvida[] = config.exclusivas
    .filter((raca) => !nomesGlobais.has(chaveDeRaca(raca.nome)))
    .map((raca) => ({
      ...raca,
      origem: "CAMPANHA",
      bloqueada: Boolean(config.overrides[raca.id]?.bloqueada),
      personalizada: false,
    }));

  return [...globais, ...exclusivas];
}

import { Fragment, useEffect, useMemo, useState, type FormEvent } from "react";
import { ATTRIBUTE_CONFIG, type BaseAttributeKey } from "../../data/personaRules";

interface CampaignRacesProps {
  campanhaId: string;
  isMaster: boolean;
  catalogo: RaceDefinition[];
  config: CampaignRaceConfig;
  /** true quando a configuração não pôde ser lida: bloqueia o salvamento para não apagar o que já existe. */
  falhaCarga: boolean;
  onSaved: (config: CampaignRaceConfig) => void;
}

type Filtro = "TODAS" | "DISPONIVEIS" | "BLOQUEADAS";

function criarId() {
  const uuid = globalThis.crypto?.randomUUID?.()
    ?? `${Date.now()}-${Math.random().toString(36).slice(2)}`;
  return `raca-campanha-${uuid}`;
}

function bonusZerado(): Record<BaseAttributeKey, number> {
  return Object.fromEntries(
    ATTRIBUTE_CONFIG.map(({ key }) => [key, 0]),
  ) as Record<BaseAttributeKey, number>;
}

function formatar(valor: number) {
  return valor > 0 ? `+${valor}` : `${valor}`;
}

function resumoAtributos(raca: RaceResolvida) {
  const partes = ATTRIBUTE_CONFIG
    .filter(({ key }) => raca.atributoBonus[key] !== 0)
    .map(({ key, abbr }) => `${abbr} ${formatar(raca.atributoBonus[key])}`);
  return partes.length > 0 ? partes.join(" · ") : "Sem bônus";
}

function limparOverrides(overrides: Record<string, CampaignRaceOverride>) {
  return Object.fromEntries(
    Object.entries(overrides).filter(
      ([, o]) =>
        o.bloqueada
        || o.atributoBonus !== undefined
        || o.hpBonus !== undefined
        || o.mpBonus !== undefined,
    ),
  );
}

export default function CampaignRaces({
  campanhaId,
  isMaster,
  catalogo,
  config,
  falhaCarga,
  onSaved,
}: CampaignRacesProps) {
  const [rascunho, setRascunho] = useState<CampaignRaceConfig>(config);
  const [filtro, setFiltro] = useState<Filtro>("TODAS");
  const [busca, setBusca] = useState("");
  const [editando, setEditando] = useState<string | null>(null);
  const [salvando, setSalvando] = useState(false);
  const [mensagem, setMensagem] = useState("");
  const [erro, setErro] = useState("");

  const [nomeNova, setNomeNova] = useState("");
  const [descricaoNova, setDescricaoNova] = useState("");
  const [bonusNova, setBonusNova] = useState<Record<BaseAttributeKey, number>>(bonusZerado);
  const [hpNova, setHpNova] = useState(0);
  const [mpNova, setMpNova] = useState(0);
  const [erroNova, setErroNova] = useState("");

  // Só volta ao que está salvo quando o servidor trouxer algo diferente
  // (recarregar a página de campanha por outro motivo não apaga edições em andamento).
  const configSalva = JSON.stringify(config);
  useEffect(() => {
    setRascunho(JSON.parse(configSalva) as CampaignRaceConfig);
  }, [configSalva]);

  const alterado = JSON.stringify(rascunho) !== configSalva;
  const podeEditar = isMaster && !falhaCarga;

  const resolvidas = useMemo(() => racasDaCampanha(catalogo, rascunho), [catalogo, rascunho]);
  const totalBloqueadas = resolvidas.filter((raca) => raca.bloqueada).length;
  const totalDisponiveis = resolvidas.length - totalBloqueadas;

  const termo = chaveDeRaca(busca);
  const visiveis = resolvidas.filter((raca) => {
    if (filtro === "DISPONIVEIS" && raca.bloqueada) return false;
    if (filtro === "BLOQUEADAS" && !raca.bloqueada) return false;
    return !termo || chaveDeRaca(raca.nome).includes(termo);
  });

  function mudarOverride(id: string, alterar: (atual: CampaignRaceOverride) => CampaignRaceOverride) {
    setMensagem("");
    setRascunho((atual) => ({
      ...atual,
      overrides: limparOverrides({
        ...atual.overrides,
        [id]: alterar(atual.overrides[id] ?? {}),
      }),
    }));
  }

  function alternarBloqueio(raca: RaceResolvida) {
    mudarOverride(raca.id, (o) => {
      const proximo = { ...o };
      if (raca.bloqueada) delete proximo.bloqueada;
      else proximo.bloqueada = true;
      return proximo;
    });
  }

  function editarBonus(raca: RaceResolvida, campo: BaseAttributeKey | "hp" | "mp", valor: number) {
    setMensagem("");
    if (raca.origem === "CAMPANHA") {
      setRascunho((atual) => ({
        ...atual,
        exclusivas: atual.exclusivas.map((item) => {
          if (item.id !== raca.id) return item;
          if (campo === "hp") return { ...item, hpBonus: valor };
          if (campo === "mp") return { ...item, mpBonus: valor };
          return { ...item, atributoBonus: { ...item.atributoBonus, [campo]: valor } };
        }),
      }));
      return;
    }
    mudarOverride(raca.id, (o) => {
      if (campo === "hp") return { ...o, hpBonus: valor };
      if (campo === "mp") return { ...o, mpBonus: valor };
      return { ...o, atributoBonus: { ...raca.atributoBonus, ...o.atributoBonus, [campo]: valor } };
    });
  }

  function restaurarPadrao(raca: RaceResolvida) {
    mudarOverride(raca.id, (o) => {
      const proximo = { ...o };
      delete proximo.atributoBonus;
      delete proximo.hpBonus;
      delete proximo.mpBonus;
      return proximo;
    });
  }

  function removerExclusiva(raca: RaceResolvida) {
    const confirmado = window.confirm(
      `Remover a raça "${raca.nome}" desta campanha?\n\nPersonagens que já usam essa raça mantêm o nome, mas deixam de receber os bônus dela.`,

    if (!confirmado) return;
    setMensagem("");
    setEditando((atual) => (atual === raca.id ? null : atual));
    setRascunho((atual) => ({
      overrides: limparOverrides(
        Object.fromEntries(Object.entries(atual.overrides).filter(([id]) => id !== raca.id)),
      ),
      exclusivas: atual.exclusivas.filter((item) => item.id !== raca.id),
    }));
  }

  function adicionarExclusiva(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();
    const nome = nomeNova.trim();
    if (!nome) return;
    if (resolvidas.some((raca) => chaveDeRaca(raca.nome) === chaveDeRaca(nome))) {

      atributoBonus: { ...bonusNova },
      hpBonus: hpNova,
      mpBonus: mpNova,
      tracos: [],
    };
    setMensagem("");
    setRascunho((atual) => ({ ...atual, exclusivas: [...atual.exclusivas, nova] }));
    setNomeNova("");
    setDescricaoNova("");
    setBonusNova(bonusZerado());
    setHpNova(0);
    setMpNova(0);
    setErroNova("");

    } catch (error) {

            >

                          onClick={() => setEditando((atual) => (atual === raca.id ? null : raca.id))}
                        >
                          {editando === raca.id ? "Fechar bônus" : "Editar bônus"}
                        </button>
                        {raca.origem === "CAMPANHA" && (
                          <button type="button" className="is-danger" onClick={() => removerExclusiva(raca)}>
                            Remover
                          </button>
                        )}
                      </div>
                    </td>
                  )}
                </tr>
                {podeEditar && editando === raca.id && (
                  <tr className="cr-edit-row">
                    <td colSpan={6}>
                      <div className="cr-bonus-grid" role="group" aria-label={`Bônus de ${raca.nome} nesta campanha`}>

                            type="number"
                            aria-label={`${raca.nome} - Vida`}
                            value={raca.hpBonus}
                            onChange={(event) => editarBonus(raca, "hp", Number(event.target.value) || 0)}
                          />
                        </label>
                        <label>
                          <span>Mana (PM)</span>
                          <input
                            type="number"
                            aria-label={`${raca.nome} - Mana`}
                            value={raca.mpBonus}

                      {raca.personalizada && (

                        </button>
                      )}

                )}
              </Fragment>
            ))}
            {visiveis.length === 0 && (
              <tr>
                <td colSpan={podeEditar ? 6 : 5} className="cr-empty">
                  Nenhuma raça encontrada com esse filtro.
                </td>
              </tr>
            )}
          </tbody>
        </table>
      </div>

      {podeEditar && (
        <form className="cr-new" onSubmit={adicionarExclusiva}>
          <h4>Criar raça exclusiva desta campanha</h4>
          <div className="cr-new-grid">
            <label className="cr-wide">

              <label key={key}>
                <span>{label}</span>
                <input
                  type="number"
                  aria-label={`Bônus de ${label} da nova raça`}
                  value={bonusNova[key]}
                  onChange={(event) =>
                    setBonusNova((atual) => ({ ...atual, [key]: Number(event.target.value) || 0 }))
                  }
                />
              </label>
            ))}
            <label>
              <span>Vida (PV)</span>
              <input
                type="number"
                aria-label="Bônus de PV da nova raça"
                value={hpNova}
                onChange={(event) => setHpNova(Number(event.target.value) || 0)}
              />
            </label>
            <label>
              <span>Mana (PM)</span>
              <input
                type="number"
                aria-label="Bônus de PM da nova raça"
                value={mpNova}
                onChange={(event) => setMpNova(Number(event.target.value) || 0)}
              />
            </label>
          </div>
          {erroNova && <p className="cr-alert" role="alert">{erroNova}</p>}
          <button type="submit" className="btn-ghost" disabled={!nomeNova.trim()}>
            Adicionar raça à campanha
          </button>
          <p className="cr-hint">A raça só passa a valer depois de clicar em “Salvar raças”.</p>
        </form>
      )}
    </div>
  );
}

function textoSeguro(valor: unknown): string {
  return typeof valor === "string" ? valor.trim().slice(0, 500) : "";
}

function numeroSeguro(valor: unknown, fallback = 0): number {
  return typeof valor === "number" && Number.isFinite(valor)
    ? Math.trunc(valor)
    : fallback;
}

const ATRIBUTOS_BASE: BaseAttributeKey[] = [
  "forca",
  "destreza",
  "constituicao",
  "inteligencia",
  "carisma",
];

export function normalizarCatalogoClasses(valor: unknown): ClassDefinition[] {
  if (!Array.isArray(valor)) return DEFAULT_CLASS_CATALOG;

  return valor.flatMap((item): ClassDefinition[] => {
    if (typeof item !== "object" || item === null) return [];
    const entrada = item as Record<string, unknown>;
    const id = textoSeguro(entrada.id);
    const nome = textoSeguro(entrada.nome);
    if (!id || !nome) return [];
    const bonusEntrada =
      typeof entrada.atributoBonus === "object" && entrada.atributoBonus !== null
        ? (entrada.atributoBonus as Record<string, unknown>)
        : {};
    const atributoBonus = Object.fromEntries(
      ATRIBUTOS_BASE.map((atributo) => [
        atributo,
        numeroSeguro(bonusEntrada[atributo]),
      ]),
    ) as Record<BaseAttributeKey, number>;
    const pericias: ClassSkillDefinition[] = Array.isArray(entrada.pericias)
      ? entrada.pericias.flatMap((itemPericia): ClassSkillDefinition[] => {
          if (typeof itemPericia !== "object" || itemPericia === null) return [];
          const pericia = itemPericia as Record<string, unknown>;
          const periciaId = textoSeguro(pericia.id);
          const nomePericia = textoSeguro(pericia.nome);
          const atributo = pericia.atributo;
          if (
            !periciaId ||
            !nomePericia ||
            typeof atributo !== "string" ||
            !ATRIBUTOS_BASE.includes(atributo as BaseAttributeKey)
          ) return [];
          return [{
            id: periciaId,
            nome: nomePericia,
            atributo: atributo as BaseAttributeKey,
            descricao: textoSeguro(pericia.descricao),
          }];
        })
      : [];
    const habilidades: ClassAbilityDefinition[] = Array.isArray(entrada.habilidades)
      ? entrada.habilidades.flatMap((itemHabilidade): ClassAbilityDefinition[] => {
          if (typeof itemHabilidade !== "object" || itemHabilidade === null) return [];
          const habilidade = itemHabilidade as Record<string, unknown>;
          const habilidadeId = textoSeguro(habilidade.id);
          const nomeHabilidade = textoSeguro(habilidade.nome);
          if (!habilidadeId || !nomeHabilidade) return [];
          return [{
            id: habilidadeId,
            nome: nomeHabilidade,
            descricao: textoSeguro(habilidade.descricao),
            nivel: Math.max(1, Math.min(20, numeroSeguro(habilidade.nivel, 1))),
          }];
        })
      : [];
    const magias: ClassSpellDefinition[] = Array.isArray(entrada.magias)
      ? entrada.magias.flatMap((itemMagia): ClassSpellDefinition[] => {
          if (typeof itemMagia !== "object" || itemMagia === null) return [];
          const magia = itemMagia as Record<string, unknown>;
          const magiaId = textoSeguro(magia.id);
          const nomeMagia = textoSeguro(magia.nome);
          if (!magiaId || !nomeMagia) return [];
          return [{
            id: magiaId,
            nome: nomeMagia,
            descricao: textoSeguro(magia.descricao),
            custo: Math.max(0, numeroSeguro(magia.custo)),
            nivel: Math.max(1, Math.min(20, numeroSeguro(magia.nivel, 1))),
          }];
        })
      : [];

    return [{
      id,
      nome,
      aliases: Array.isArray(entrada.aliases)
        ? entrada.aliases.map(textoSeguro).filter(Boolean)
        : [],
      usaMagia: entrada.usaMagia === true,
      atributoBonus,
      hpBonus: numeroSeguro(entrada.hpBonus),
      mpBonus: numeroSeguro(entrada.mpBonus),
      pericias,
      habilidades,
      magias,
    }];
  });
}

export function normalizarBonusClassesCampanha(
  valores?: unknown,
  incluirPadroes = true,
): CampaignClassBonuses {
  const entrada =
    typeof valores === "object" && valores !== null
      ? (valores as Record<string, unknown>)
      : {};
  const resultado: CampaignClassBonuses = {};

  for (const [classe, definicao] of incluirPadroes
    ? Object.entries(CLASS_BONUSES)
    : []) {
    const personalizado =
      typeof entrada[classe] === "object" && entrada[classe] !== null
        ? (entrada[classe] as Record<string, unknown>)
        : {};
    const atributosPersonalizados =
      typeof personalizado.atributoBonus === "object" &&
      personalizado.atributoBonus !== null
        ? (personalizado.atributoBonus as Record<string, unknown>)
        : {};
    const atributoBonus: Record<BaseAttributeKey, number> = {
      forca: definicao.atributoBonus?.forca ?? 0,
      destreza: definicao.atributoBonus?.destreza ?? 0,
      constituicao: definicao.atributoBonus?.constituicao ?? 0,
      inteligencia: definicao.atributoBonus?.inteligencia ?? 0,
      carisma: definicao.atributoBonus?.carisma ?? 0,
    };

    for (const atributo of Object.keys(atributoBonus) as BaseAttributeKey[]) {
      const valor = atributosPersonalizados[atributo];
      if (typeof valor === "number" && Number.isFinite(valor)) {
        atributoBonus[atributo] = Math.trunc(valor);
      }
    }

    resultado[classe] = {
      atributoBonus,
      hpBonus:
        typeof personalizado.hpBonus === "number" &&
        Number.isFinite(personalizado.hpBonus)
          ? Math.trunc(personalizado.hpBonus)
          : definicao.hpBonus ?? 0,
      mpBonus:
        typeof personalizado.mpBonus === "number" &&
        Number.isFinite(personalizado.mpBonus)
          ? Math.trunc(personalizado.mpBonus)
          : definicao.mpBonus ?? 0,
      bloqueada: Boolean(personalizado.bloqueada),
    };
  }

  for (const [classe, entradaClasse] of Object.entries(entrada)) {
    if (classe in resultado || typeof entradaClasse !== "object" || entradaClasse === null) {
      continue;
    }
    const personalizado = entradaClasse as Record<string, unknown>;
    const atributos =
      typeof personalizado.atributoBonus === "object" && personalizado.atributoBonus !== null
        ? (personalizado.atributoBonus as Record<string, unknown>)
        : {};
    const atributoBonus = Object.fromEntries(
      ATRIBUTOS_BASE.map((atributo) => [
        atributo,
        numeroSeguro(atributos[atributo]),
      ]),
    ) as Record<BaseAttributeKey, number>;
    resultado[classe] = {
      atributoBonus,
      hpBonus: numeroSeguro(personalizado.hpBonus),
      mpBonus: numeroSeguro(personalizado.mpBonus),
      bloqueada: Boolean(personalizado.bloqueada),
    };
  }

  return resultado;
}

create or replace function public._integer_from_jsonb(
  p_value jsonb,
  p_fallback integer,
  p_min integer,
  p_max integer
)
returns integer
language plpgsql
immutable
set search_path = pg_catalog, public
as $$
declare
  v_value numeric;
begin
  if p_value is null or jsonb_typeof(p_value) is distinct from 'number' then
    return p_fallback;
  end if;

  v_value := (p_value #>> '{}')::numeric;
  return greatest(p_min, least(p_max, trunc(v_value)))::integer;
exception
  when numeric_value_out_of_range or invalid_text_representation then
    return p_fallback;
end;
$$;

      or exists (
        select 1
        from jsonb_array_elements_text(
          case when jsonb_typeof(entrada.value -> 'aliases') = 'array'
            then entrada.value -> 'aliases' else '[]'::jsonb end
        ) as class_alias(nome)
        where public._character_rule_key(class_alias.nome) = v_chave_classe
      )
    )

    v_tem_classe_global := found;

    v_bonus_classe := v_override;

    -- or does not contain this class, even when other global classes exist.
    v_bonus_classe := public._builtin_character_bonus('CLASSE', v_classe);
  end if;

  v_bonus_atributo := coalesce(v_bonus_raca -> 'atributoBonus', '{}'::jsonb);
  v_forca := public._integer_from_jsonb(p_dados #> '{attributes,forca}', 10, 1, 30)

  v_constituicao := public._integer_from_jsonb(p_dados #> '{attributes,constituicao}', 10, 1, 30)
    + public._integer_from_jsonb(v_bonus_atributo -> 'constituicao', 0, -999999, 999999);
  v_inteligencia := public._integer_from_jsonb(p_dados #> '{attributes,inteligencia}', 10, 1, 30)
    + public._integer_from_jsonb(v_bonus_atributo -> 'inteligencia', 0, -999999, 999999);
  v_carisma := public._integer_from_jsonb(p_dados #> '{attributes,carisma}', 10, 1, 30)
    + public._integer_from_jsonb(v_bonus_atributo -> 'carisma', 0, -999999, 999999);

  v_bonus_atributo := coalesce(v_bonus_classe -> 'atributoBonus', '{}'::jsonb);
  v_forca := v_forca + public._integer_from_jsonb(v_bonus_atributo -> 'forca', 0, -999999, 999999);
  v_destreza := v_destreza + public._integer_from_jsonb(v_bonus_atributo -> 'destreza', 0, -999999, 999999);
  v_constituicao := v_constituicao + public._integer_from_jsonb(v_bonus_atributo -> 'constituicao', 0, -999999, 999999);
  v_inteligencia := v_inteligencia + public._integer_from_jsonb(v_bonus_atributo -> 'inteligencia', 0, -999999, 999999);
  v_carisma := v_carisma + public._integer_from_jsonb(v_bonus_atributo -> 'carisma', 0, -999999, 999999);

  v_bonus_atributo := coalesce(v_bonus_filiacao -> 'atributoBonus', '{}'::jsonb);
  v_forca := v_forca + public._integer_from_jsonb(v_bonus_atributo -> 'forca', 0, -999999, 999999);

    when 'Guerreiro' then 10
    when 'Paladino' then 10
    when 'Patrulheiro' then 10

    when 'Ladino' then 8
    when 'Monge' then 8
    when 'Feiticeiro' then 6
    when 'Mago' then 6
    else 8

  v_por_nivel := greatest(1, floor(v_dado_vida / 2.0) + 1 + v_modificador_con);
  v_hp := greatest(
    1,
    least(
      200,
      v_dado_vida + v_modificador_con + (v_nivel - 1) * v_por_nivel

      + public._integer_from_jsonb(v_bonus_filiacao -> 'mpBonus', 0, -999999, 999999)
    )
  );

  return query select v_hp::integer, v_mp::integer;
end;

create or replace function public.aplicar_limites_recursos_personagem()
returns trigger
language plpgsql
security definer
set search_path = pg_catalog, public, pg_temp
as $$
declare
  v_hp_max integer;
  v_mp_max integer;
  v_hp jsonb;
  v_mp jsonb;
begin
  if new.dados is null or jsonb_typeof(new.dados) is distinct from 'object' then
    return new;
  end if;

  select maxima.hp_max, maxima.mp_max
    into v_hp_max, v_mp_max
  from public._calcular_maximos_recursos_personagem(
    new.id,
    new.dados,
    new.nivel,
    new.campanha_id
  ) maxima;

  v_hp := case
    when jsonb_typeof(new.dados -> 'hp') = 'object' then new.dados -> 'hp'
    else '{}'::jsonb
  end;
  v_mp := case
    when jsonb_typeof(new.dados -> 'mp') = 'object' then new.dados -> 'mp'
    else '{}'::jsonb
  end;
  v_hp := jsonb_set(
    v_hp - 'max',
    '{atual}',
    to_jsonb(public._integer_from_jsonb(v_hp -> 'atual', 0, 0, v_hp_max)),
    true
  );
  v_mp := jsonb_set(
    v_mp - 'max',
    '{atual}',
    to_jsonb(public._integer_from_jsonb(v_mp -> 'atual', 0, 0, v_mp_max)),
    true
  );
  new.dados := jsonb_set(
    jsonb_set(new.dados, '{hp}', v_hp, true),
    '{mp}',
    v_mp,
    true
  );
  return new;
end;
$$;

create or replace function public.usar_item_personagem(
  p_personagem_id uuid,
  p_item_id uuid,
  p_hp_max integer default null,
  p_mp_max integer default null
)
returns jsonb
language plpgsql
security definer
set search_path = pg_catalog, public, pg_temp
as $$
declare
  v_usuario_id uuid := auth.uid();
  v_personagem public.personagens%rowtype;
  v_inventario public.inventarios%rowtype;
  v_item public.itens%rowtype;
  v_campanha_id uuid;
  v_dados jsonb;
  v_recurso jsonb;
  v_cargas integer;
  v_quantidade integer;
  v_novo_valor integer;
  v_maximo integer;
  v_hp_max integer;
  v_mp_max integer;
  v_delta integer;
begin
  if v_usuario_id is null then
    raise exception 'Usuário não autenticado.';
  end if;

  select * into v_personagem
  from public.personagens
  where id = p_personagem_id
  for update;
  if not found then
    raise exception 'Personagem não encontrado.';
  end if;
  v_campanha_id := v_personagem.campanha_id;

  if v_personagem.user_id is distinct from v_usuario_id and not exists (
    select 1 from public.campanha_membros membro
    where membro.campanha_id = v_campanha_id
      and membro.user_id = v_usuario_id
      and membro.papel = 'MESTRE'
      and membro.status = 'ATIVO'
  ) then
    raise exception 'Você não pode usar itens deste personagem.';
  end if;

  -- Serialize equipped-item changes against the authoritative equipment bonuses.
  perform inventario.item_id
  from public.inventarios inventario
  where inventario.personagem_id = p_personagem_id
  order by inventario.item_id
  for update;

  select * into v_inventario
  from public.inventarios
  where personagem_id = p_personagem_id and item_id = p_item_id
  for update;
  if not found or v_inventario.quantidade <= 0 then
    raise exception 'Item não encontrado no inventário.';
  end if;

  select * into v_item from public.itens where id = p_item_id;
  if not found then
    raise exception 'Definição do item não encontrada.';
  end if;
  if v_item.tipo is distinct from 'CONSUMIVEL' and v_item.cargas_maximas is null then
    raise exception 'Este item não pode ser consumido.';
  end if;

  select maxima.hp_max, maxima.mp_max
    into v_hp_max, v_mp_max
  from public._calcular_maximos_recursos_personagem(
    p_personagem_id,
    coalesce(v_personagem.dados, '{}'::jsonb),
    v_personagem.nivel,
    v_campanha_id
  ) maxima;

  v_dados := coalesce(v_personagem.dados, '{}'::jsonb);
  v_delta := public._integer_from_jsonb(
    v_item.efeito_mecanico -> 'valor', 0, -999999, 999999
  );
  if v_item.efeito_mecanico ->> 'tipo' = 'RECURSO' then
    if v_item.efeito_mecanico ->> 'recurso' not in ('hp', 'mp') then
      raise exception 'O recurso deste item é inválido.';
    end if;

    if v_item.efeito_mecanico ->> 'recurso' = 'hp' then
      v_recurso := case
        when jsonb_typeof(v_dados -> 'hp') = 'object' then v_dados -> 'hp'
        else '{}'::jsonb
      end;
      v_maximo := v_hp_max;
    else
      v_recurso := case
        when jsonb_typeof(v_dados -> 'mp') = 'object' then v_dados -> 'mp'
        else '{}'::jsonb
      end;
      v_maximo := v_mp_max;
    end if;

    v_novo_valor := greatest(
      0,
      least(
        v_maximo,
        public._integer_from_jsonb(v_recurso -> 'atual', 0, 0, 200)
          + greatest(v_delta, 0)
      )
    );
    v_recurso := jsonb_set(v_recurso, '{atual}', to_jsonb(v_novo_valor), true);
    if v_item.efeito_mecanico ->> 'recurso' = 'hp' then
      v_dados := jsonb_set(v_dados, '{hp}', v_recurso, true);
    else
      v_dados := jsonb_set(v_dados, '{mp}', v_recurso, true);
    end if;
  end if;

  if v_item.cargas_maximas is not null then
    v_cargas := coalesce(
      v_inventario.cargas_atuais,
      v_item.cargas_maximas * v_inventario.quantidade
    );
    if v_cargas <= 0 then
      raise exception 'Este item está sem cargas.';
    end if;
    v_cargas := v_cargas - 1;
    v_quantidade := least(
      v_inventario.quantidade,
      ceil(v_cargas::numeric / v_item.cargas_maximas)::integer
    );
    if v_quantidade <= 0 then
      delete from public.inventarios
      where personagem_id = p_personagem_id and item_id = p_item_id;
    else
      update public.inventarios
      set quantidade = v_quantidade,
          cargas_atuais = v_cargas,
          updated_at = now()
      where personagem_id = p_personagem_id and item_id = p_item_id;
    end if;
  elsif v_item.tipo = 'CONSUMIVEL' then
    v_quantidade := v_inventario.quantidade - 1;
    if v_quantidade <= 0 then
      delete from public.inventarios
      where personagem_id = p_personagem_id and item_id = p_item_id;
    else
      update public.inventarios
      set quantidade = v_quantidade,
          updated_at = now()
      where personagem_id = p_personagem_id and item_id = p_item_id;
    end if;
  end if;

  update public.personagens
  set dados = v_dados
  where id = p_personagem_id;

  return jsonb_build_object(
    'personagem_id', p_personagem_id,
    'hp', v_dados -> 'hp',
    'mp', v_dados -> 'mp',
    'quantidade', greatest(0, v_quantidade),
    'hp_max', v_hp_max,
    'mp_max', v_mp_max
  );
end;
$$;

$$;

  p_personagem_id uuid,
  p_dados jsonb,
  p_nivel integer,
  p_campanha_id uuid
)
returns table (hp_max integer, mp_max integer)
language plpgsql
security invoker
set search_path = pg_catalog, public
as $$
declare
  v_classe text := coalesce(p_dados ->> 'classe', 'Aventureiro');
  v_classe_id text := p_dados ->> 'classeId';
  v_raca text := coalesce(p_dados ->> 'raca', '');
  v_filiacao text := coalesce(p_dados ->> 'filiacao', '');
  v_bonus_raca jsonb;
  v_bonus_classe jsonb;
  v_bonus_filiacao jsonb;
  v_override jsonb;
  v_conteudo_campanha jsonb;
  v_classe_global jsonb;
  v_catalogo_global jsonb;
  v_bonus_campanha jsonb := '{}'::jsonb;
  v_tem_override boolean := false;
  v_tem_conteudo_campanha boolean := false;
  v_tem_catalogo_global boolean := false;
  v_tem_classe_global boolean := false;
  v_chave_classe text := public._character_rule_key(v_classe);
  v_forca numeric;
  v_destreza numeric;
  v_constituicao numeric;
  v_inteligencia numeric;
  v_carisma numeric;
  v_bonus_equip_constituicao numeric := 0;
  v_bonus_equip_inteligencia numeric := 0;
  v_bonus_atributo jsonb;
  v_dado_vida integer;
  v_nivel integer := greatest(1, coalesce(p_nivel, 1));
  v_modificador_con numeric;
  v_por_nivel numeric;
  v_hp numeric;
  v_mp numeric;
begin
  v_bonus_raca := public._builtin_character_bonus('RACA', v_raca);
  v_bonus_filiacao := public._builtin_character_bonus('FILIACAO', v_filiacao);

  if p_campanha_id is not null then
    select coalesce(campanha.bonus_classes, '{}'::jsonb)
      into v_bonus_campanha
    from public.campanhas campanha
    where campanha.id = p_campanha_id
    for share;

    select entrada.value
      into v_override
    from jsonb_each(
      case when jsonb_typeof(v_bonus_campanha) = 'object'
        then v_bonus_campanha else '{}'::jsonb end
    ) as entrada(key, value)
    where jsonb_typeof(entrada.value) = 'object'
      and public._character_rule_key(entrada.key) in (
        v_chave_classe,
        public._character_rule_key(v_classe_id)
      )
    order by case
      when public._character_rule_key(entrada.key) = public._character_rule_key(v_classe_id) then 0
      else 1
    end
    limit 1;
    v_tem_override := found;

    select jsonb_build_object(
      'atributoBonus', conteudo.bonus_atributos,
      'hpBonus', conteudo.bonus_hp,
      'mpBonus', conteudo.bonus_mp
    )
      into v_conteudo_campanha
    from public.campanha_classes_conteudos conteudo
    where conteudo.campanha_id = p_campanha_id
      and conteudo.tipo = 'CLASSE'
      and (
        (v_classe_id is not null and conteudo.classe_id = v_classe_id)
        or public._character_rule_key(conteudo.nome) = v_chave_classe
      )
    order by conteudo.created_at, conteudo.id
    limit 1;
    v_tem_conteudo_campanha := found;
  end if;

  select catalogo.dados
    into v_catalogo_global
  from public.global_class_catalog catalogo
  where catalogo.id = 'global'
  for share;
  v_tem_catalogo_global := found;

  if v_tem_catalogo_global then
    select entrada.value
      into v_classe_global
    from jsonb_array_elements(
      case when jsonb_typeof(v_catalogo_global) = 'array'
        then v_catalogo_global else '[]'::jsonb end
    ) with ordinality as entrada(value, ordem)
    where (
      (v_classe_id is not null and entrada.value ->> 'id' = v_classe_id)
      or public._character_rule_key(entrada.value ->> 'nome') = v_chave_classe
      or exists (
        select 1
        from jsonb_array_elements_text(
          case when jsonb_typeof(entrada.value -> 'aliases') = 'array'
            then entrada.value -> 'aliases' else '[]'::jsonb end
        ) as class_alias(nome)
        where public._character_rule_key(class_alias.nome) = v_chave_classe
      )
    )
    order by entrada.ordem
    limit 1;
    v_tem_classe_global := found;
  end if;

  if v_tem_override then
    v_bonus_classe := v_override;
  elsif v_tem_conteudo_campanha then
    v_bonus_classe := v_conteudo_campanha;
  elsif v_tem_classe_global then
    v_bonus_classe := v_classe_global;
  else
    v_bonus_classe := public._builtin_character_bonus('CLASSE', v_classe);
  end if;

  v_bonus_atributo := coalesce(v_bonus_raca -> 'atributoBonus', '{}'::jsonb);
  v_forca := public._integer_from_jsonb(p_dados #> '{attributes,forca}', 10, 1, 30)
    + public._integer_from_jsonb(v_bonus_atributo -> 'forca', 0, -999999, 999999);
  v_destreza := public._integer_from_jsonb(p_dados #> '{attributes,destreza}', 10, 1, 30)
    + public._integer_from_jsonb(v_bonus_atributo -> 'destreza', 0, -999999, 999999);
  v_constituicao := public._integer_from_jsonb(p_dados #> '{attributes,constituicao}', 10, 1, 30)
    + public._integer_from_jsonb(v_bonus_atributo -> 'constituicao', 0, -999999, 999999);
  v_inteligencia := public._integer_from_jsonb(p_dados #> '{attributes,inteligencia}', 10, 1, 30)
    + public._integer_from_jsonb(v_bonus_atributo -> 'inteligencia', 0, -999999, 999999);
  v_carisma := public._integer_from_jsonb(p_dados #> '{attributes,carisma}', 10, 1, 30)
    + public._integer_from_jsonb(v_bonus_atributo -> 'carisma', 0, -999999, 999999);

  v_bonus_atributo := coalesce(v_bonus_classe -> 'atributoBonus', '{}'::jsonb);
  v_forca := v_forca + public._integer_from_jsonb(v_bonus_atributo -> 'forca', 0, -999999, 999999);
  v_destreza := v_destreza + public._integer_from_jsonb(v_bonus_atributo -> 'destreza', 0, -999999, 999999);
  v_constituicao := v_constituicao + public._integer_from_jsonb(v_bonus_atributo -> 'constituicao', 0, -999999, 999999);
  v_inteligencia := v_inteligencia + public._integer_from_jsonb(v_bonus_atributo -> 'inteligencia', 0, -999999, 999999);
  v_carisma := v_carisma + public._integer_from_jsonb(v_bonus_atributo -> 'carisma', 0, -999999, 999999);

  v_bonus_atributo := coalesce(v_bonus_filiacao -> 'atributoBonus', '{}'::jsonb);
  v_forca := v_forca + public._integer_from_jsonb(v_bonus_atributo -> 'forca', 0, -999999, 999999);
  v_destreza := v_destreza + public._integer_from_jsonb(v_bonus_atributo -> 'destreza', 0, -999999, 999999);
  v_constituicao := v_constituicao + public._integer_from_jsonb(v_bonus_atributo -> 'constituicao', 0, -999999, 999999);
  v_inteligencia := v_inteligencia + public._integer_from_jsonb(v_bonus_atributo -> 'inteligencia', 0, -999999, 999999);
  v_carisma := v_carisma + public._integer_from_jsonb(v_bonus_atributo -> 'carisma', 0, -999999, 999999);

  select
    coalesce(sum(public._integer_from_jsonb(item.bonus_atributos -> 'constituicao', 0, -999999, 999999)), 0),
    coalesce(sum(public._integer_from_jsonb(item.bonus_atributos -> 'inteligencia', 0, -999999, 999999)), 0)
    into v_bonus_equip_constituicao, v_bonus_equip_inteligencia
  from public.inventarios inventario
  join public.itens item on item.id = inventario.item_id
  where inventario.personagem_id = p_personagem_id
    and inventario.equipado = true
    and inventario.quantidade > 0;

  v_constituicao := v_constituicao + v_bonus_equip_constituicao;
  v_inteligencia := v_inteligencia + v_bonus_equip_inteligencia;

  v_bonus_raca := coalesce(v_bonus_raca, '{}'::jsonb);
  v_bonus_filiacao := coalesce(v_bonus_filiacao, '{}'::jsonb);
  v_bonus_classe := coalesce(v_bonus_classe, '{}'::jsonb);

  v_dado_vida := case v_classe
    when 'Bárbaro' then 12
    when 'Guerreiro' then 10
    when 'Paladino' then 10
    when 'Patrulheiro' then 10
    when 'Bardo' then 8
    when 'Bruxo' then 8
    when 'Clérigo' then 8
    when 'Druida' then 8
    when 'Ladino' then 8
    when 'Monge' then 8
    when 'Feiticeiro' then 6
    when 'Mago' then 6
    else 8
  end;

  v_modificador_con := floor((v_constituicao - 10) / 2);
  v_por_nivel := greatest(1, floor(v_dado_vida / 2.0) + 1 + v_modificador_con);
  v_hp := greatest(
    1,
    least(
      200,
      v_dado_vida + v_modificador_con + (v_nivel - 1) * v_por_nivel
      + public._integer_from_jsonb(v_bonus_raca -> 'hpBonus', 0, -999999, 999999)
      + public._integer_from_jsonb(v_bonus_classe -> 'hpBonus', 0, -999999, 999999)
      + public._integer_from_jsonb(v_bonus_filiacao -> 'hpBonus', 0, -999999, 999999)
    )
  );
  v_mp := greatest(
    0,
    least(
      200,
      8 + v_inteligencia
      + public._integer_from_jsonb(v_bonus_raca -> 'mpBonus', 0, -999999, 999999)
      + public._integer_from_jsonb(v_bonus_classe -> 'mpBonus', 0, -999999, 999999)
      + public._integer_from_jsonb(v_bonus_filiacao -> 'mpBonus', 0, -999999, 999999)
    )
  );

  return query select v_hp::integer, v_mp::integer;
end;
$$;

create or replace function public.aplicar_limites_recursos_personagem()
returns trigger
language plpgsql
security definer
set search_path = pg_catalog, public, pg_temp
as $$
declare
  v_hp_max integer;
  v_mp_max integer;
  v_hp jsonb;
  v_mp jsonb;
begin
  if new.dados is null or jsonb_typeof(new.dados) is distinct from 'object' then
    return new;
  end if;

  select maxima.hp_max, maxima.mp_max
    into v_hp_max, v_mp_max
  from public._calcular_maximos_recursos_personagem(
    new.id,
    new.dados,
    new.nivel,
    new.campanha_id
  ) maxima;

  v_hp := case
    when jsonb_typeof(new.dados -> 'hp') = 'object' then new.dados -> 'hp'
    else '{}'::jsonb
  end;
  v_mp := case
    when jsonb_typeof(new.dados -> 'mp') = 'object' then new.dados -> 'mp'
    else '{}'::jsonb
  end;
  v_hp := jsonb_set(
    v_hp - 'max',
    '{atual}',
    to_jsonb(public._integer_from_jsonb(v_hp -> 'atual', 0, 0, v_hp_max)),
    true
  );
  v_mp := jsonb_set(
    v_mp - 'max',
    '{atual}',
    to_jsonb(public._integer_from_jsonb(v_mp -> 'atual', 0, 0, v_mp_max)),
    true
  );
  new.dados := jsonb_set(
    jsonb_set(new.dados, '{hp}', v_hp, true),
    '{mp}',
    v_mp,
    true
  );
  return new;
end;
$$;

create or replace function public.incrementar_versao_dados_personagem()
returns trigger
language plpgsql
set search_path = public, pg_temp
as $$
begin
  if row(new.dados, new.nome, new.nivel, new.campanha_id)
     is distinct from row(old.dados, old.nome, old.nivel, old.campanha_id) then
    new.dados_version := old.dados_version + 1;
  else
    new.dados_version := old.dados_version;
  end if;
  return new;
end;
$$;

create or replace function public.creditar_ouro_campanha(
  p_campanha_id uuid,
  p_quantia integer,
  p_personagem_id uuid default null,
  p_request_id uuid default gen_random_uuid()
)
returns integer
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  v_usuario_id uuid := auth.uid();
  v_request_id uuid := coalesce(p_request_id, gen_random_uuid());
  v_operation_id bigint;
  v_existing public.campanha_recompensas%rowtype;
  v_personagem record;
  v_dados jsonb;
  v_carteira jsonb;
  v_novo_saldo numeric;
  v_atualizados integer := 0;
begin
  if v_usuario_id is null then
    raise exception 'Usuário não autenticado.';
  end if;
  if p_quantia is null or p_quantia <= 0 or p_quantia > 100000 then
    raise exception 'A quantia deve estar entre 1 e 100000 PO.';
  end if;
  if not exists (
    select 1 from public.campanha_membros membro
    where membro.campanha_id = p_campanha_id
      and membro.user_id = v_usuario_id
      and membro.papel = 'MESTRE'
      and membro.status = 'ATIVO'
  ) then
    raise exception 'Somente o Mestre ativo pode distribuir ouro.';
  end if;

  insert into public.campanha_recompensas (
    campanha_id, personagem_id, tipo, quantidade, request_id, created_by
  ) values (
    p_campanha_id, p_personagem_id, 'OURO', p_quantia, v_request_id, v_usuario_id
  ) on conflict (created_by, request_id) do nothing
  returning id into v_operation_id;

  if v_operation_id is null then
    select * into v_existing
    from public.campanha_recompensas
    where created_by = v_usuario_id and request_id = v_request_id;
    if not found
      or v_existing.campanha_id <> p_campanha_id
      or v_existing.personagem_id is distinct from p_personagem_id
      or v_existing.tipo <> 'OURO'
      or v_existing.item_id is not null
      or v_existing.quantidade <> p_quantia then
      raise exception 'A chave da recompensa já foi usada com outros dados.';
    end if;
    return v_existing.total_destinatarios;
  end if;

  for v_personagem in
    select personagem.id, personagem.dados
    from public.personagens personagem
    join public.campanha_membros membro
      on membro.campanha_id = personagem.campanha_id
      and membro.user_id = personagem.user_id
      and membro.status = 'ATIVO'
    where personagem.campanha_id = p_campanha_id
      and (p_personagem_id is null or personagem.id = p_personagem_id)
    order by personagem.id
    for update
  loop
    v_dados := coalesce(v_personagem.dados, '{}'::jsonb);
    v_carteira := case
      when jsonb_typeof(v_dados -> 'carteira') = 'object' then v_dados -> 'carteira'
      else '{}'::jsonb
    end;
    v_novo_saldo := coalesce(nullif(v_carteira ->> 'po', '')::numeric, 0) + p_quantia;
    if v_novo_saldo > 9999999 then
      raise exception 'O saldo de % excederia o limite de 9999999 PO.', v_personagem.id;
    end if;
    v_carteira := v_carteira || jsonb_build_object('po', v_novo_saldo);
    v_dados := jsonb_set(v_dados, '{carteira}', v_carteira, true);
    update public.personagens set dados = v_dados where id = v_personagem.id;
    v_atualizados := v_atualizados + 1;
  end loop;

  if p_personagem_id is not null and v_atualizados = 0 then
    raise exception 'Personagem não pertence a esta campanha.';
  end if;
  update public.campanha_recompensas
  set total_destinatarios = v_atualizados
  where id = v_operation_id;
  return v_atualizados;
end;
$$;

create or replace function public.creditar_experiencia_campanha(
  p_campanha_id uuid,
  p_quantia integer,
  p_personagem_id uuid default null,
  p_request_id uuid default gen_random_uuid()
)
returns integer
language plpgsql
security definer
set search_path = public, pg_temp
as $$
declare
  v_usuario_id uuid := auth.uid();
  v_request_id uuid := coalesce(p_request_id, gen_random_uuid());

begin
  if v_usuario_id is null then
    raise exception 'Usuário não autenticado.';
  end if;
  if p_quantia is null or p_quantia <= 0 or p_quantia > 2147483647 then
    raise exception 'A quantidade de XP deve ser maior que zero.';
  end if;
  if not exists (
    select 1 from public.campanha_membros membro
    where membro.campanha_id = p_campanha_id
      and membro.user_id = v_usuario_id
      and membro.papel = 'MESTRE'
      and membro.status = 'ATIVO'
  ) then
    raise exception 'Somente o Mestre ativo pode distribuir XP.';
  end if;

  insert into public.campanha_recompensas (
    campanha_id, personagem_id, tipo, quantidade, request_id, created_by
  ) values (
    p_campanha_id, p_personagem_id, 'XP', p_quantia, v_request_id, v_usuario_id
  ) on conflict (created_by, request_id) do nothing
  returning id into v_operation_id;

  if v_operation_id is null then
    select * into v_existing
    from public.campanha_recompensas
    where created_by = v_usuario_id and request_id = v_request_id;
    if not found
      or v_existing.campanha_id <> p_campanha_id
      or v_existing.personagem_id is distinct from p_personagem_id
      or v_existing.tipo <> 'XP'
      or v_existing.item_id is not null
      or v_existing.quantidade <> p_quantia then
      raise exception 'A chave da recompensa já foi usada com outros dados.';
    end if;
    return v_existing.total_destinatarios;
  end if;

  for v_personagem in
    select personagem.id, personagem.xp
    from public.personagens personagem
    join public.campanha_membros membro
      on membro.campanha_id = personagem.campanha_id
      and membro.user_id = personagem.user_id
      and membro.status = 'ATIVO'
    where personagem.campanha_id = p_campanha_id
      and (p_personagem_id is null or personagem.id = p_personagem_id)
    order by personagem.id
    for update
  loop
    v_novo_xp := coalesce(v_personagem.xp, 0)::bigint + p_quantia;
    if v_novo_xp > 2147483647 then
      raise exception 'O XP excederia o limite armazenável para o personagem %.', v_personagem.id;
    end if;
    update public.personagens
    set xp = v_novo_xp::integer
    where id = v_personagem.id;
    v_atualizados := v_atualizados + 1;
  end loop;

  if p_personagem_id is not null and v_atualizados = 0 then
    raise exception 'Personagem não pertence a esta campanha.';
  end if;
  update public.campanha_recompensas
  set total_destinatarios = v_atualizados
  where id = v_operation_id;
  return v_atualizados;
end;
$$;

create or replace function public.entregar_item_campanha(
  p_campanha_id uuid,
  p_item_id uuid,
  p_quantidade integer,

as $$
declare
  v_usuario_id uuid := auth.uid();
  v_request_id uuid := coalesce(p_request_id, gen_random_uuid());
  v_operation_id bigint;
  v_existing public.campanha_recompensas%rowtype;
  v_personagem record;
  v_updated integer;
  v_destinatarios integer := 0;
begin
  if v_usuario_id is null then
    raise exception 'Usuário não autenticado.';
  end if;
  if p_quantidade is null or p_quantidade <= 0 or p_quantidade > 9999 then
    raise exception 'A quantidade de itens deve estar entre 1 e 9999.';
  end if;
  if not exists (select 1 from public.itens item where item.id = p_item_id) then
    raise exception 'O item selecionado não existe mais.';
  end if;
  if not exists (
    select 1 from public.campanha_membros membro
    where membro.campanha_id = p_campanha_id
      and membro.user_id = v_usuario_id
      and membro.papel = 'MESTRE'
      and membro.status = 'ATIVO'
  ) then
    raise exception 'Somente o Mestre ativo pode entregar itens.';
  end if;

  insert into public.campanha_recompensas (
    campanha_id, personagem_id, item_id, tipo, quantidade, quantidade_item, request_id, created_by
  ) values (
    p_campanha_id, p_personagem_id, p_item_id, 'ITEM', p_quantidade, p_quantidade, v_request_id, v_usuario_id
  ) on conflict (created_by, request_id) do nothing
  returning id into v_operation_id;

  if v_operation_id is null then
    select * into v_existing
    from public.campanha_recompensas
    where created_by = v_usuario_id and request_id = v_request_id;
    if not found
      or v_existing.campanha_id <> p_campanha_id
      or v_existing.personagem_id is distinct from p_personagem_id
      or v_existing.tipo <> 'ITEM'
      or v_existing.item_id is distinct from p_item_id
      or v_existing.quantidade_item <> p_quantidade then
      raise exception 'A chave da recompensa já foi usada com outros dados.';
    end if;
    return v_existing.total_destinatarios;
  end if;

  for v_personagem in
    select personagem.id
    from public.personagens personagem
    join public.campanha_membros membro
      on membro.campanha_id = personagem.campanha_id
      and membro.user_id = personagem.user_id
      and membro.status = 'ATIVO'
    where personagem.campanha_id = p_campanha_id
      and (p_personagem_id is null or personagem.id = p_personagem_id)
    order by personagem.id
    for update
  loop
    update public.inventarios
      set quantidade = quantidade + p_quantidade,
          updated_at = now()
      where personagem_id = v_personagem.id
        and item_id = p_item_id;
    get diagnostics v_updated = row_count;
    if v_updated = 0 then
      insert into public.inventarios (personagem_id, item_id, quantidade, equipado, updated_at)
      values (v_personagem.id, p_item_id, p_quantidade, false, now());
    end if;
    v_destinatarios := v_destinatarios + 1;
  end loop;

  if p_personagem_id is not null and v_destinatarios = 0 then
    raise exception 'Personagem não pertence a esta campanha.';
  end if;
  update public.campanha_recompensas
  set total_destinatarios = v_destinatarios
  where id = v_operation_id;
  return v_destinatarios;
end;
$$;

create or replace function public.listar_maximos_recursos_campanha(p_campanha_id uuid)
returns table (
  personagem_id uuid,
  hp_atual integer,
  hp_max integer,
  mp_atual integer,
  mp_max integer
)
language plpgsql
security definer
set search_path = public, pg_temp
as $$
begin
  if auth.uid() is null or not exists (
    select 1 from public.campanha_membros membro
    where membro.campanha_id = p_campanha_id
      and membro.user_id = auth.uid()
      and membro.status = 'ATIVO'
  ) then
    raise exception 'Você não participa ativamente desta campanha.';
  end if;

  return query
    select
      personagem.id,
      public._integer_from_jsonb(personagem.dados #> '{hp,atual}', 0, 0, maxima.hp_max),
      maxima.hp_max,
      public._integer_from_jsonb(personagem.dados #> '{mp,atual}', 0, 0, maxima.mp_max),
      maxima.mp_max
    from public.personagens personagem
    join public.campanha_membros membro
      on membro.campanha_id = personagem.campanha_id
      and membro.user_id = personagem.user_id
      and membro.status = 'ATIVO'
    cross join lateral public._calcular_maximos_recursos_personagem(
      personagem.id, personagem.dados, personagem.nivel, personagem.campanha_id
    ) maxima
    where personagem.campanha_id = p_campanha_id
    order by personagem.id;
end;
$$;
